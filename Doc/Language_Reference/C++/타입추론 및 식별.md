# 타입추론 및 식별：Type Inference & Identification

## 타입 추론

&nbsp; C++ 는 선언된 시점을 추적하여 해당 변수의 타입을 추론할 수 있습니다. auto 나 decltype 을 사용하여 이 기능을 능동적으로 사용할 수 있습니다.

개선된 타입 추론(auto, decltype(), 리턴 타입)

auto는 템플릿 클래스 추론를 추론
decltype는 선언된 그대로 추론

-   decltype은 레퍼런스도 추론 가능하기 때문에 레퍼런스를 리턴할 수 있게 된다.
-   -> decltype(auto)를 사용해도 무방하다.

## 타입 식별 코드

-   type_id().name 을 통하여 타입 이름을 가져온 뒤, abi 인터페이스로 demangle 하면 내부 구현된 상태를 알 수 있습니다.
-   더 깔끔하게 보고 싶다면 내부에 구분용으로 구현된 이름을 리네이밍하는 작업을 거치면 됩니다.

```cpp
#include <bits/stdc++.h>
#include <cxxabi.h>
// └ abi::__cxa_demangle

using namespace std;

template <typename T>
string getType(const T& target, const bool& isRename = true) {
    const char* mangledName = typeid(T).name();
    int status = 0;

    // * demangle 압축된 타입 이름 해제
    unique_ptr<char, void(*)(void*)> demangled(abi::__cxa_demangle(mangledName, nullptr, nullptr, &status), free);

    string typeName = (status == 0 && demangled) ? string(demangled.get()) : string (mangledName);

    // * GCC 내부 C++11 ABI(Application Binary Interface) 내부구현이름을 리네이밍
    if (isRename) {

        // ** 전부 replace 해버리는 함수 람다식으로 정의 <= 수정 필요
        auto replaceAll = [](string& str, const string& from, const string& to, const string& brk = "") -> void {
            size_t start = 0;
            size_t end = 0;
            while ((start = str.find(from, start)) != string::npos) {
                end = start + from.size();

                if (brk != "") {
                    if ((end + 1 < str.size()) && (str[end] == brk[0])) {
                        int depth = 0;
                        while (end < str.size()) {
                            if (str[end] == brk[0]) depth++;
                            else if (str[end] == brk[1]) depth--;
                            end++;
                            if (depth == 0) break;
                        }
                    }
                }

                str.replace(start, end - start, to);
                start += to.length();
            }
        };

        // ** alias 타입 명칭 변경 ================================
        // *** "std::basic_string<wchar_t>" -> "std::wstring"
        replaceAll(typeName, "basic_string<wchar_t, std::char_traits<wchar_t>, std::allocator<wchar_t> >", "wstring");

        // *** "std::basic_string<char>" -> "std::string"
        replaceAll(typeName, "basic_string<char, std::char_traits<char>, std::allocator<char> >", "string");

        // ** 템플릿인자 제거 =====================================
        // *** ", std::__cxx11::<...>" -> ""
        replaceAll(typeName, "__cxx11::", "");

        // *** ", std::less<...>" -> ""
        replaceAll(typeName, ", std::less", "", "<>");

        // *** ", std::hash<...>" -> ""
        replaceAll(typeName, ", std::hash", "", "<>");

        // *** ", std::equal_to<...>" -> ""
        replaceAll(typeName, ", std::equal_to", "", "<>");

        // *** ", std::allocator<...>" -> ""
        replaceAll(typeName, ", std::allocator", "", "<>");

        // ** 괄호 띄어쓰기 조정 ==================================
        replaceAll(typeName, " >", ">");
    }

    return typeName;
}

int main() {
    cout << "[TEST CODE]" << "\n";

    cout << " * [RENAME OFF] " << getType(string("string"), false) << "\n";
    cout << " * [RENAME  ON] " << getType(string("string")) << "\n";
    cout << "\n";

    vector<string> sv(10);
    cout << " * [RENAME OFF] " << getType(sv, false) << "\n";
    cout << " * [RENAME  ON] " << getType(sv) << "\n";
    cout << "\n";

    vector<int> iv(10);
    cout << " * [RENAME OFF] " << getType(iv.begin(), false) << "\n";
    cout << " * [RENAME  ON] " << getType(iv.begin()) << "\n";
    cout << "\n";

    map<string, int> m;
    cout << " * [RENAME OFF] " << getType(m, false) << "\n";
    cout << " * [RENAME  ON] " << getType(m) << "\n";
    cout << "\n";

    unordered_map<long long, float> um;
    cout << " * [RENAME OFF] " << getType(um, false) << "\n";
    cout << " * [RENAME  ON] " << getType(um) << "\n";
    cout << "\n";

    set<wstring> s;
    cout << " * [RENAME OFF] " << getType(s, false) << "\n";
    cout << " * [RENAME  ON] " << getType(s) << "\n";
    cout << "\n";

    return 0;
}
```
