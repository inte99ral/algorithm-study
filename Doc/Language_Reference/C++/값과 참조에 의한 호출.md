### 맵의 포인터를 받는 컨테이너

```cpp
#include <bits/stdc++.h>
using namespace std;
int main() {

    // !  `deque<pair<const int, int>&> dq;` 은 불가능.
    //    컨테이너는 참조 타입을 원소로 가질 수 없음.

    deque<pair<const int, int>*> dq;
    map<int, int> m;
    m[1] = 5;
    m[2] = 6;

    for (auto& i : m) dq.push_back(&i);

    cout << dq.front()->first << "-" << m[dq.front()->first] << "\n";
    dq.front()->second = 10;
    cout << dq.front()->first << "-" << m[dq.front()->first] << "\n";

    return 0;
}
```

/**
 * 값에 의한 호출(Call by value) 와 참조에 의한 호출(Call by reference)
 * 또는 깊은 복사(Deep Copy)와 얕은 복사(Shallow Copy)
 */
#include <iostream>
#include <vector>

using namespace std;

/**
 * 참조에 의한 호출, 또는 얕은 복사
 * 새로운 메모리 공간에 값을 완전히 복사하여 다룬다.
 * 원본 데이터에 엑세스 할 수 없기 때문에 변경에 안전하다.
 * 통체로 복사되는 만큼 생성될 때마다 연산 비용과 저장 메모리를 감안해야 함.
 */
void callByRefVec(vector<int>* v) {
    for (int i = 0; i < (*v).size(); i++) {
        cout << (*v)[i] << ' ';
        (*v)[i] = 0;
    }
    cout << endl;
}

void callByRefArr(int arr[3][3]) {
    for (int y = 0; y < 3; y++) {
        for (int x = 0; x < 3; x++) {
            cout << arr[y][x] << ' ';
        }
    }
    cout << endl;
}

/**

void callByValVec(vector<int> v) {

}
