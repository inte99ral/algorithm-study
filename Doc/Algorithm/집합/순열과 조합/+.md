# 순열과 조합 실전

## 순열

```c++
#include <bits/stdc++.h>
using namespace std;
int N = 0;
int R = 0;
vector<int> ORIGIN = vector<int>();

void permRecur(int size) {
    if (size == R){
        for (int i = 0; i < R; i++) cout << ORIGIN[i] << ", ";
        cout << "\b\b \n";
        return;
    }
    for (int i = size; i < N; i++) {
        swap(ORIGIN[i], ORIGIN[size]);
        permRecur(size + 1);
        swap(ORIGIN[i], ORIGIN[size]);
    }
    return;
}

void perm(int n, int r, vector<int> origin) {
    N = n;
    R = r;
    ORIGIN = origin;
    permRecur(0);
    return;
}

int main() {
    perm(3, 2, {10, 20, 30});
    return 0;
}
```

## 조합

```c++
#include <bits/stdc++.h>
using namespace std;
int N = 0;
int R = 0;
vector<int> ORIGIN = vector<int>();
deque<int> SELECT;

void combRecur(int curr) {
    if ((int) SELECT.size() == R) {
        for (int i : SELECT) cout << i << ", ";
        cout << "\b\b \n";
        return;
    }
    for (int i = curr; i < N; i++) {
        SELECT.push_back(ORIGIN[i]);
        combRecur(i + 1);
        SELECT.pop_back();
    }
    return;
}

void comb(int n, int r, vector<int> origin) {
    N = n;
    R = r;
    ORIGIN = origin;
    combRecur(0);
    return;
}

int main(int argc, char* argv[]) {
    comb(3, 2, {10, 20, 30});
    return 0;
}
```
