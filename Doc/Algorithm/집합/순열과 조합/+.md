## 순열

### 순열 실전지향형 재귀 Swap

```cpp
#include <bits/stdc++.h>
using namespace std;

int N;
int R;
vector<int> origin;
// vector<int> select; 불필요

// * permutation : depth 깊이의 원소를 뽑는 알고리즘
void perm(int depth) {
    // * depth 깊이가 출력을 의도한 R 만큼 되면 그만 뽑고 재귀 종료하고 출력하기
    if (depth == R) {
        for (int r = 0; r < R; r++) cout << origin[r] << ", ";
        cout << "\b\b \n";
        return;
    }

    // * "depth 이상으로 큰 순서의" origin 원소를 n 위치와 교환
    for (int n = depth; n < N; n++) {
        swap(origin[depth], origin[n]);
        perm(depth + 1);
        swap(origin[depth], origin[n]);
    }
}

int main() {
    N = 4;
    R = 2;
    origin = vector<int>(N);
    for (int n = 0; n < N; n++) origin[n] = n;

    perm(0);
    return 0;
}
```

### 순열 실전지향형 재귀 방문검색 백트래킹

```cpp
#include <bits/stdc++.h>
using namespace std;

int N;
int R;
vector<int> origin;
vector<int> select;
vector<bool> visited; // * 방문기록 필요

// * permutation : depth 깊이의 원소를 뽑는 알고리즘
void perm(int depth) {
    // * depth 깊이가 출력을 의도한 R 만큼 되면 그만 뽑고 재귀 종료하고 출력하기
    if (depth == R) {
        for (int i : select) cout << i << ", ";
        cout << "\b\b \n";
        return;
    }

    // * "visited 한 적 없는" origin 원소를 select 의 depth 위치에 넣기
    for (int n = 0; n < N; n++) {
        if (visited[n]) continue;

        visited[n] = 1;
        select[depth] = origin[n];
        perm(depth + 1);
        visited[n] = 0;
    }
}

int main() {
    N = 4;
    R = 2;
    origin = vector<int>(N);
    select = vector<int>(R);
    visited = vector<bool>(N);
    for (int n = 0; n < N; n++) origin[n] = n;

    perm(0);
    return 0;
}
```

### 순열 실전지향형 재귀 반복자

```c++
#include <bits/stdc++.h>
using namespace std;

int N;
int R;
vector<int> origin;
// vector<int> select; 불필요
// vector<bool> visited; 불필요

int main() {
    N = 4;
    R = 2;
    origin = vector<int>(N);
    for (int n = 0; n < N; n++) origin[n] = n;

    do {
        // 사전적 오름차순 -> 내림차순 과정을 밟으며 0~R 까지의 원소만 확인하면 됨
        for (int r = 0; r < R; r++) cout << origin[r] << ", ";
        cout << "\b\b \n";

        // r번째 이후의 요소들을 내림차순으로 뒤집어 과정 넘겨버리기
        reverse(origin.begin() + R, origin.end());
    } while(next_permutation(origin.begin(), origin.end()));
    
    return 0;
}
```

## 조합

### 조합 실전지향형 재귀

```cpp
#include <bits/stdc++.h>
using namespace std;

int N;
int R;
vector<int> origin;
vector<int> select;

// * combination : depth 깊이의 원소를 뽑는 알고리즘
void comb(int depth) {
    // * depth 깊이가 출력을 의도한 R 만큼 되면 그만 뽑고 재귀 종료하고 출력하기
    if (depth == R) {
        for (int r = 0; r < R; r++) cout << select[r] << ' ';
        cout << '\n';
        return;
    }

    // * "이전의 선택보다 +1 큰 순서의" origin 원소를 select 의 depth 위치에 넣기
    for (int n = (depth? select[depth - 1] + 1 : 0); n < N; n++) {
        select[depth] = origin[n];
        comb(depth + 1);
        // select[depth] = 0; 어차피 덮어씌워지므로 원상복구 불필요
    }
    return;
}

int main() {
    N = 6;
    R = 3;
    origin = vector<int>(N);
    select = vector<int>(R);
    for (int n = 0; n < N; n++) origin[n] = n;

    comb(0);
    return 0;
}
```

### 조합 실전지향형 재귀 반복자

```cpp
#include <bits/stdc++.h>
using namespace std;

int N;
int R;
vector<int> origin;
vector<bool> select; // * 마스크 {선택 | 비선택}
// vector<bool> visited; 불필요

int main() {
    N = 4;
    R = 2;
    origin = vector<int>(N);
    select = vector<bool>(N);

    for (int n = 0; n < N; n++) origin[n] = n;
    for (int r = 0; r < R; r++) select[r] = 1;

    do {
        // {1, 1, 0, 0} -> {0, 0, 1, 1}
        // 사전적 내림차순 -> 오름차순 과정을 밟으며 select 선택된 원소만 출력
        for (int n = 0; n < N; n++) if (select[n]) cout << origin[n] << ", ";
        cout << "\b\b \n";

        // reverse(); 이미 r 개 선택제한은 select 배열 1 원소 갯수로 결정됨
    } while(prev_permutation(select.begin(), select.end()));
    return 0;
}
```
