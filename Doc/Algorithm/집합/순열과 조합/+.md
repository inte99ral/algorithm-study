# 순열과 조합 실전

## 순열

```c++
#include <bits/stdc++.h>
using namespace std;

bool isSolved = false;
int n = 6;
int r = 2;
vector<int> origin = { 1, 2, 3, 4, 5, 6 };

void perm(int o_idx) {
    if (o_idx == r) {
        for (int i = 0; i < r; i++) {
            cout << origin[i] << ", ";
        }
        cout << "\b\b \n";
        return;
    }
    for (int i = o_idx; i < n; i++) {
        swap(origin[o_idx], origin[i]);
        perm(o_idx + 1);
        swap(origin[o_idx], origin[i]);
    }
    return;
}

int main() {
    perm(0);
    return 0;
}
```

## 조합

```c++
#include <bits/stdc++.h>
using namespace std;
int N = 0;
int R = 0;
vector<int> ORIGIN = vector<int>();
deque<int> SELECT;

void combRecur(int curr) {
    if ((int) SELECT.size() == R) {
        for (int i : SELECT) cout << i << ", ";
        cout << "\b\b \n";
        return;
    }
    for (int i = curr; i < N; i++) {
        SELECT.push_back(ORIGIN[i]);
        combRecur(i + 1);
        SELECT.pop_back();
    }
    return;
}

void comb(int n, int r, vector<int> origin) {
    N = n;
    R = r;
    ORIGIN = origin;
    combRecur(0);
    return;
}

int main(int argc, char* argv[]) {
    comb(3, 2, {10, 20, 30});
    return 0;
}
```
