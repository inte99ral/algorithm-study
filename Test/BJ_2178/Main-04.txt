// * 메모리 초과
#include <bits/stdc++.h>
using namespace std;

int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(nullptr);
    cout.tie(nullptr);

    // freopen("_INPUT_.txt", "rt", stdin); // <======================================

    int ans, cnt;
    int N, M;
    vector<vector<bool>> matrix;
    deque<pair<int, int>> task;

    cin >> N >> M;
    matrix = vector<vector<bool>>(N, vector<bool>(M));
    for (int n = 0; n < N; n++) {
        string temp;
        cin >> temp;
        for (int m = 0; m < M; m++) matrix[n][m] = (temp.at(m) == '1');
    }

    // 카운트 생성 -> 전부소진하면 ans+1
    task.emplace_back(0, 0);
    ans = 1;
    cnt = 1;

    while(!task.empty()) {
        int n = task.front().first;
        int m = task.front().second;
        task.pop_front();
        matrix[n][m] = 0;

        if ((n == N - 1) && (m == M - 1)) break;

        if((n > 0) && matrix[n - 1][m]) task.emplace_back(n - 1, m);
        if((n < N - 1) && matrix[n + 1][m]) task.emplace_back(n + 1, m);
        if((m > 0) && matrix[n][m - 1]) task.emplace_back(n, m - 1);
        if((m < M - 1) && matrix[n][m + 1]) task.emplace_back(n, m + 1);
        
        cnt--;
        if (cnt == 0) {
            ans++;
            cnt = task.size();
        } 
    }

    cout << ans;

    return 0;
}
