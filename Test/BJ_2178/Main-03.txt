#include <bits/stdc++.h>
using namespace std;

int N;
int R;
vector<int> origin;
vector<int> select;

void comb1(int depth) {
    if (depth == R) {
        for (int i : select) cout << i << ' ';
        cout << '\n';
        return;
    }
    for (int n = (depth? select[depth - 1] + 1 : 0); n < N; n++) {
        select[depth] = origin[n];
        comb1(depth + 1);
    }
    return;
}

void comb(int depth) {
    // * depth 깊이가 출력을 의도한 R 만큼 되면 그만 뽑고 재귀 종료하고 출력하기
    if (depth == R) {
        for (int r = 0; r < R; r++) cout << select[r] << ' ';
        cout << '\n';
        return;
    }

    // * "이전의 선택보다 +1 큰 순서의" origin 원소를 select 의 depth 위치에 넣기
    for (int n = (depth? select[depth - 1] + 1 : 0); n < N; n++) {
        select[depth] = origin[n];
        comb(depth + 1);
        // select[depth] = 0; 어차피 덮어씌워지므로 원상복구 불필요
    }
    return;
}

int main() {
    N = 6;
    R = 3;
    origin = vector<int>(N);
    select = vector<int>(R);
    for (int n = 0; n < N; n++) origin[n] = n + 1;

    comb(0);
    return 0;
}
